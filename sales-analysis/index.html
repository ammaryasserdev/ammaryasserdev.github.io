أداة تحليل المبيعات بـ Python: الإيراد وأفضل شهر وترتيب المنتجات

Python / Pandas / Matplotlib

سكربت بايثون يقرأ ملف مبيعات CSV ويجيب عن أسئلة حقيقية لصاحب العمل: كم الإيراد الكلي؟ ما أفضل شهر وأسوأ شهر؟ وأي المنتجات تجلب أكبر دخل؟ ثم يرسم رسمين بيانيين ويكتب ملخص نتائج قصيرًا. يعمل على أي ملف بنفس الأعمدة.

رسم توضيحي للفكرة، وليس لقطة من الأداة.

الفكرة

ماذا يفعل السكربت؟

يقرأ السكربت ملف المبيعات، ويتحقق من البيانات، ثم يحسب الإيراد لكل صف (الكمية × سعر الوحدة) ويجمّعه حسب الشهر وحسب المنتج. النتيجة: رسمان بيانيان وملف ملخص نصي جاهز للقراءة.

نفس روح الأداة الأولى: لا صندوق أسود. الصفوف التي فيها قيم ناقصة أو سالبة تُستبعد ويُكتب عددها في الملخص، ولا يتم تخمين أي رقم.

القدرات

ما الذي يجيب عنه؟

  1. إجمالي الإيراد عن الفترة كاملة.
  2. أفضل شهر وأسوأ شهر من حيث الإيراد، مع متوسط الشهر.
  3. ترتيب المنتجات حسب الإيراد مع نسبة كل منتج من الإجمالي.
  4. رسمان بيانيان بصيغة PNG: اتجاه الإيراد الشهري، والإيراد حسب المنتج.
  5. ملخص نتائج وقرارات موثّق يُكتب ويُطبع مع كل تشغيل.

الكود المصدري الكامل

"""
Sales Analysis - a small, explainable sales-analysis tool.

What it does, step by step:
  1. Loads a sales CSV (columns: order_id, order_date, product, quantity, unit_price).
  2. Checks the data (bad dates, missing numbers, negative values) and reports what it skipped.
  3. Computes revenue = quantity * unit_price for every row.
  4. Answers the business questions:
       - What is the total revenue?
       - Which month was the best, and which was the worst?
       - How do the products rank by revenue?
  5. Saves two charts (monthly revenue trend, revenue by product).
  6. Writes a short findings summary to a text file and prints it.

Works on any CSV with the same columns.

Usage:
  python sales_analysis.py sales.csv
  python sales_analysis.py sales.csv --out-dir results
"""

import argparse
import os
import sys

import matplotlib
matplotlib.use("Agg")  # draw charts to files, no window needed
import matplotlib.pyplot as plt
import pandas as pd

REQUIRED = ["order_id", "order_date", "product", "quantity", "unit_price"]


def load_and_check(path, notes):
    """Load the CSV and keep only rows we can trust. Every skipped row is recorded in notes."""
    df = pd.read_csv(path)
    missing = [c for c in REQUIRED if c not in df.columns]
    if missing:
        sys.exit("Missing required column(s): " + ", ".join(missing))
    notes.append(f"Loaded {len(df)} rows from {os.path.basename(path)}.")

    df["order_date"] = pd.to_datetime(df["order_date"], errors="coerce")
    df["quantity"] = pd.to_numeric(df["quantity"], errors="coerce")
    df["unit_price"] = pd.to_numeric(df["unit_price"], errors="coerce")

    bad = df[REQUIRED].isna().any(axis=1)
    if bad.any():
        notes.append(f"Skipped {int(bad.sum())} row(s) with a missing or unreadable value.")
        df = df[~bad]

    negative = (df["quantity"] <= 0) | (df["unit_price"] <= 0)
    if negative.any():
        notes.append(f"Skipped {int(negative.sum())} row(s) with zero or negative quantity/price "
                     "(returns should be reviewed separately).")
        df = df[~negative]

    df = df.copy()
    df["revenue"] = df["quantity"] * df["unit_price"]
    df["month"] = df["order_date"].dt.to_period("M")
    return df


def analyse(df):
    """Return the three answers: total, monthly revenue, product revenue."""
    total = df["revenue"].sum()
    monthly = df.groupby("month")["revenue"].sum().sort_index()
    products = df.groupby("product")["revenue"].sum().sort_values(ascending=False)
    return total, monthly, products


def save_charts(monthly, products, out_dir):
    labels = [str(m) for m in monthly.index]
    fig, ax = plt.subplots(figsize=(8, 4.2))
    ax.plot(labels, monthly.values, marker="o", color="#73a89a", linewidth=2.5)
    ax.fill_between(labels, monthly.values, color="#73a89a", alpha=0.15)
    ax.set_title("Monthly revenue")
    ax.set_ylabel("Revenue (EGP)")
    ax.grid(axis="y", alpha=0.3)
    fig.tight_layout()
    fig.savefig(os.path.join(out_dir, "monthly_revenue.png"), dpi=120)
    plt.close(fig)

    fig, ax = plt.subplots(figsize=(8, 4.2))
    ordered = products.sort_values()
    ax.barh(ordered.index, ordered.values, color="#73a89a")
    ax.set_title("Revenue by product")
    ax.set_xlabel("Revenue (EGP)")
    ax.grid(axis="x", alpha=0.3)
    fig.tight_layout()
    fig.savefig(os.path.join(out_dir, "revenue_by_product.png"), dpi=120)
    plt.close(fig)


def write_summary(df, total, monthly, products, notes, path):
    best, worst = monthly.idxmax(), monthly.idxmin()
    top_share = products.iloc[0] / total * 100
    lines = ["Sales findings", "=" * 14]
    lines += notes
    lines += [
        f"Period: {df['order_date'].min().date()} to {df['order_date'].max().date()} "
        f"({df['order_id'].nunique()} orders).",
        f"Total revenue: {total:,.0f} EGP.",
        f"Best month: {best} with {monthly[best]:,.0f} EGP.",
        f"Worst month: {worst} with {monthly[worst]:,.0f} EGP.",
        f"Average per month: {monthly.mean():,.0f} EGP.",
        "",
        "Product ranking by revenue:",
    ]
    for i, (name, value) in enumerate(products.items(), 1):
        lines.append(f"  {i}. {name}: {value:,.0f} EGP ({value / total * 100:.1f}%)")
    lines += [
        "",
        f"Decision: the top product ({products.index[0]}) brings {top_share:.1f}% of revenue, "
        f"and the lowest ({products.index[-1]}) brings {products.iloc[-1] / total * 100:.1f}%.",
        "Decision: revenue is quantity x unit_price; rows with bad values are skipped, not guessed.",
    ]
    text = "\n".join(lines)
    with open(path, "w", encoding="utf-8") as f:
        f.write(text + "\n")
    return text


def main():
    parser = argparse.ArgumentParser(description="Analyse a sales CSV.")
    parser.add_argument("input", help="path to the sales CSV")
    parser.add_argument("--out-dir", default=".", help="folder for charts and summary")
    args = parser.parse_args()

    os.makedirs(args.out_dir, exist_ok=True)
    notes = []
    df = load_and_check(args.input, notes)
    if df.empty:
        sys.exit("No usable rows after checking the data.")
    total, monthly, products = analyse(df)
    save_charts(monthly, products, args.out_dir)
    print(write_summary(df, total, monthly, products, notes,
                        os.path.join(args.out_dir, "findings.txt")))


if __name__ == "__main__":
    main()
